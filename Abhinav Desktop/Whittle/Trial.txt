Survival probabilities: p[x] = Pr(survive from state x)
Higher state = lower survival probability (monotone decreasing)
p_survive = np.array([0.9, 0.7, 0.4, 0.1])

============================================================================
STEP 2: Construct Extended State Space (x, a_prev)
============================================================================
Extended state: (x, a_prev) where x ∈ {0,1,2,3}, a_prev ∈ {0,1}
a_prev=1: active (repairman was assigned last period)
a_prev=0: passive (no repairman last period)
Map extended state to index
def state_to_idx(x, a_prev):
"""Convert (x, a_prev) to linear index"""
return x * 2 + a_prev

def idx_to_state(idx):
"""Convert linear index to (x, a_prev)"""
return idx // 2, idx % 2

N_EXTENDED_STATES = N_STATES * 2 # 8 states total
print(f"Extended state space size: {N_EXTENDED_STATES}")
print(f"States: (x, a_prev) with x ∈ {{0,1,2,3}}, a_prev ∈ {{0,1}}")
print()

P0 = np.zeros((N_EXTENDED_STATES, N_EXTENDED_STATES))
P1 = np.zeros((N_EXTENDED_STATES, N_EXTENDED_STATES))
R0 = np.zeros(N_EXTENDED_STATES)
R1 = np.zeros(N_EXTENDED_STATES)

for idx in range(N_EXTENDED_STATES):
x, a_prev = idx_to_state(idx)

text
# --------- PASSIVE ACTION (a=0) ---------
# Transition: either deteriorate (x → x+1, a_next=0) or fail (x → 0, a_next=0)
if x < N_STATES - 1:
    # Survive: transition to state (x+1, 0)
    next_idx_survive = state_to_idx(x + 1, 0)
    P0[idx, next_idx_survive] = p_survive[x]

# Fail: transition to state (0, 0) with cost
next_idx_fail = state_to_idx(0, 0)
P0[idx, next_idx_fail] += (1 - p_survive[x])

# Passive cost: expected failure cost
R0[idx] = -K_FAILURE * (1 - p_survive[x])

# --------- ACTIVE ACTION (a=1) ---------
# Deterministic repair: (x, a_prev) → (0, 1)
next_idx_repair = state_to_idx(0, 1)
P1[idx, next_idx_repair] = 1.0

# Active cost depends on previous action
if a_prev == 1:
    # No switching cost
    R1[idx] = -C_REPAIR
else:  # a_prev == 0
    # Switching cost applied
    R1[idx] = -(C_REPAIR + C_SWITCH)

print(f"R1 shape (Active): {R1.shape}")

STEP 5: Closed-Form Computation via Value Iteration

def value_iteration_with_intervention_charge(P0, P1, R0, R1, w_val, discount, tol=1e-10, max_iter=10000):
"""
Compute value functions V(s; w) for a given intervention charge w.
Uses value iteration with policy evaluation.
"""
V = np.zeros(N_EXTENDED_STATES)

text
for iteration in range(max_iter):
    V_old = V.copy()
    
    for idx in range(N_EXTENDED_STATES):
        # Q-value for passive action
        Q_passive = R0[idx] + discount * np.dot(P0[idx, :], V_old)
        
        # Q-value for active action with intervention charge w
        Q_active = (R1[idx] - w_val) + discount * np.dot(P1[idx, :], V_old)
        
        # Bellman optimality: take the minimum (cost minimization)
        V[idx] = min(Q_passive, Q_active)
    
    # Check convergence
    if np.max(np.abs(V - V_old)) < tol:
        print(f"  Value iteration converged in {iteration} iterations")
        break

return V
def compute_whittle_indices_closed_form(P0, P1, R0, R1, discount, w_min=-10, w_max=50, resolution=1000):
"""
Compute Whittle indices by finding the indifference points.

text
For each state, the Whittle index W*(s) is the value of w where:
    min(Q_passive(s), Q_active(s, w)) switches from passive to active
"""
W_indices = np.zeros(N_EXTENDED_STATES)

w_values = np.linspace(w_min, w_max, resolution)

for idx in range(N_EXTENDED_STATES):
    x, a_prev = idx_to_state(idx)
    
    # For each intervention charge w, compute the optimal value
    advantages = []
    
    for w_val in w_values:
        V = value_iteration_with_intervention_charge(P0, P1, R0, R1, w_val, discount, max_iter=1000)
        V_passive = R0[idx] + discount * np.dot(P0[idx, :], V)
        V_active = (R1[idx] - w_val) + discount * np.dot(P1[idx, :], V)
        
        # Advantage: difference between passive and active Q-values
        advantage = V_passive - V_active
        advantages.append(advantage)
    
    # Find w where advantage crosses zero (indifference point)
    advantages = np.array(advantages)
    sign_changes = np.where(np.diff(np.sign(advantages)))[0]
    
    if len(sign_changes) > 0:
        idx_cross = sign_changes[0]
        w1, w2 = w_values[idx_cross], w_values[idx_cross + 1]
        adv1, adv2 = advantages[idx_cross], advantages[idx_cross + 1]
        
        # Linear interpolation for the crossing point
        if adv2 != adv1:
            W_indices[idx] = w1 + (w2 - w1) * (0 - adv1) / (adv2 - adv1)
        else:
            W_indices[idx] = w1
    else:
        # If no crossing, take the boundary
        if advantages[0] < 0:
            W_indices[idx] = w_min
        else:
            W_indices[idx] = w_max

return W_indices
