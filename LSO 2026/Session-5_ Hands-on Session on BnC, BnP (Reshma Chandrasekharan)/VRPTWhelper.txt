def route_load(route, q):

def compute_schedule(route, coord, a, b, s):
    T = {}

    # Step 1: Start service at the first node
    first_node = route[0]
    T[first_node] = max(0.0, a[first_node])

    # Step 2: Move through the route one arc at a time
    for idx in range(1, len(route)):

        # Previous node and current node
        prev = route[idx - 1]
        cur = route[idx]

        # Step 3: Compute arrival time at current node
        travel_time = distance(coord, prev, cur)
        arrival_time = T[prev] + s[prev] + travel_time

        # Step 4: If vehicle arrives early, it waits
        if arrival_time < a[cur]:
            service_start_time = a[cur]
        else:
            service_start_time = arrival_time

        T[cur] = service_start_time

        # Step 5: Check latest service time
        if T[cur] > b[cur]:
            return False, T

    # Step 6: If all nodes satisfy time windows, route is feasible
    return True, T

def route_cost(route, coord):
    total_cost = 0

    # Go through the route one arc at a time
    for i in range(len(route) - 1):

        # Current node
        current_node = route[i]

        # Next node in the route
        next_node = route[i + 1]

        # Add distance from current node to next node
        total_cost = total_cost + distance(coord, current_node, next_node)

    return total_cost

def route_is_feasible(route, Q, coord, q, a, b, s):
    # Check capacity
    load = route_load(route, q)

    if load > Q:
        return False

    # Check time windows
    feasible, T = compute_schedule(route, coord, a, b, s)

    if feasible == False:
        return False

    return True

def relocate_local_search(routes, Q, coord, q, a, b, s):
    improved = True

    while improved == True:

        improved = False

        # Try moving a customer from one route to another route
        for r1_index in range(len(routes)):

            for r2_index in range(len(routes)):

                # Do not move within the same route in this simple version
                if r1_index == r2_index:
                    continue

                route1 = routes[r1_index]
                route2 = routes[r2_index]

                old_cost = route_cost(route1, coord) + route_cost(route2, coord)

                # Try removing one customer from route1
                # Position 0 is start depot and last position is end depot
                for remove_pos in range(1, len(route1) - 1):

                    customer = route1[remove_pos]

                    # Remove customer from route1
                    new_route1 = []

                    for pos in range(len(route1)):
                        if pos != remove_pos:
                            new_route1.append(route1[pos])

                    # Try inserting this customer into every position of route2
                    # insert_pos = 1 means after start depot
                    # insert_pos = len(route2)-1 means before end depot
                    for insert_pos in range(1, len(route2)):

                        new_route2 = []

                        # Copy nodes before insertion position
                        for pos in range(insert_pos):
                            new_route2.append(route2[pos])

                        # Insert the customer
                        new_route2.append(customer)
